cmake_minimum_required(VERSION 3.24) # numbers: supported CMake API baseline.
include(cmake/Version.cmake)
project(PaperEngine VERSION "${PAPER_VERSION_CORE}" LANGUAGES C CXX)
option(PAPER_BUILD_EDITOR "Build the Qt Widgets editor" OFF)
if(APPLE AND PAPER_BUILD_EDITOR)
    enable_language(OBJCXX)
endif()
option(PAPER_BUILD_TESTS "Build engine test executables (does not run them)" OFF)
option(PAPER_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
add_library(paper_core INTERFACE)
add_library(Paper::Core ALIAS paper_core)
target_compile_features(paper_core INTERFACE cxx_std_20)
target_include_directories(paper_core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(PAPER_ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "PAPER_ENABLE_SANITIZERS requires Clang or GCC")
    endif()
    target_compile_options(paper_core INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(paper_core INTERFACE -fsanitize=address,undefined)
endif()
function(paper_target target)
    target_link_libraries(${target} PUBLIC Paper::Core)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8) # numbers: MSVC UTF-8 source encoding flag.
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wnon-virtual-dtor)
    endif()
endfunction()
